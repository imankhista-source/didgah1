/* ================================================== USER ACCOUNT SYSTEM ================================================== */ /* CHECK LOGIN WHEN APP STARTS */ document.addEventListener("DOMContentLoaded",function(){ const rememberedUser = localStorage.getItem("didgah_remembered_user"); const sessionUser = sessionStorage.getItem("didgah_session_user"); if(rememberedUser || sessionUser){ const username = rememberedUser || sessionUser; enterApp(username); }else{ showLogin(); } }); /* SHOW LOGIN */ function showLogin(){ document .getElementById("loginHeader") .classList.remove("hidden"); document .getElementById("mainHeader") .classList.add("hidden"); document .getElementById("accountMain") .classList.remove("hidden"); document .getElementById("pollHomeMain") .classList.add("hidden"); document .getElementById("loginForm") .classList.remove("hidden"); document .getElementById("registerForm") .classList.add("hidden"); closeAccountMenu(); } /* SHOW REGISTER */ function showRegister(){ document .getElementById("loginForm") .classList.add("hidden"); document .getElementById("registerForm") .classList.remove("hidden"); } /* LOGIN */ function loginUser(){ const username = document .getElementById("loginUsername") .value .trim(); const password = document .getElementById("loginPassword") .value; const remember = document .getElementById("rememberMe") .checked; const message = document .getElementById("loginMessage"); if(!username || !password){ message.innerText = "لطفاً نام کاربری و رمز عبور را وارد کنید."; return; } const savedUsername = localStorage.getItem( "didgah_registered_username" ); const savedPassword = localStorage.getItem( "didgah_registered_password" ); if( username !== savedUsername || password !== savedPassword ){ message.innerText = "نام کاربری یا رمز عبور صحیح نیست."; return; } if(remember){ localStorage.setItem( "didgah_remembered_user", username ); sessionStorage.removeItem( "didgah_session_user" ); }else{ sessionStorage.setItem( "didgah_session_user", username ); localStorage.removeItem( "didgah_remembered_user" ); } enterApp(username); } /* REGISTER */ function registerUser(){ const username = document .getElementById("registerUsername") .value .trim(); const password = document .getElementById("registerPassword") .value; const password2 = document .getElementById("registerPassword2") .value; const remember = document .getElementById("registerRemember") .checked; const message = document .getElementById("registerMessage"); if(!username || !password || !password2){ message.innerText = "لطفاً همه قسمت‌ها را کامل کنید."; return; } if(password.length < 6){ message.innerText = "رمز عبور باید حداقل ۶ کاراکتر باشد."; return; } if(password !== password2){ message.innerText = "تکرار رمز عبور با رمز عبور یکسان نیست."; return; } /* SAVE ACCOUNT FOR PROTOTYPE */ localStorage.setItem( "didgah_registered_username", username ); localStorage.setItem( "didgah_registered_password", password ); if(remember){ localStorage.setItem( "didgah_remembered_user", username ); sessionStorage.removeItem( "didgah_session_user" ); }else{ sessionStorage.setItem( "didgah_session_user", username ); localStorage.removeItem( "didgah_remembered_user" ); } enterApp(username); } /* ENTER APP */ function enterApp(username){ document .getElementById("loginHeader") .classList.add("hidden"); document .getElementById("accountMain") .classList.add("hidden"); document .getElementById("mainHeader") .classList.remove("hidden"); document .getElementById("pollHomeMain") .classList.remove("hidden"); document .getElementById("homePage") .classList.add("active"); document .getElementById("categoryPage") .classList.remove("active"); document .getElementById("pollPage") .classList.remove("active"); closeAccountMenu(); } /* ACCOUNT MENU */ function toggleAccountMenu(){ const menu = document.getElementById("accountMenu"); menu.classList.toggle("show"); } /* CLOSE ACCOUNT MENU */ function closeAccountMenu(){ const menu = document.getElementById("accountMenu"); if(menu){ menu.classList.remove("show"); } } /* LOGOUT */ function logoutUser(){ localStorage.removeItem( "didgah_remembered_user" ); sessionStorage.removeItem( "didgah_session_user" ); /* رمز حساب را فعلاً حذف نمی‌کنیم تا کاربر بعداً بتواند دوباره با همان حساب وارد شود. */ document .getElementById("loginUsername") .value=""; document .getElementById("loginPassword") .value=""; document .getElementById("loginMessage") .innerText=""; showLogin(); } /* FORGOT PASSWORD */ function forgotPassword(){ alert( "در نسخه فعلی آزمایشی، بازیابی رمز عبور هنوز به سرور متصل نشده است. این قسمت را در مرحله اتصال به سیستم واقعی حساب کاربری فعال می‌کنیم." ); } /* PROTECT CATEGORY */ const originalOpenCategory = openCategory; openCategory=function(category){ if( !localStorage.getItem("didgah_remembered_user") && !sessionStorage.getItem("didgah_session_user") ){ showLogin(); return; } originalOpenCategory(category); }; /* PROTECT POLL */ const originalOpenPoll = openPoll; openPoll=function(index){ if( !localStorage.getItem("didgah_remembered_user") && !sessionStorage.getItem("didgah_session_user") ){ showLogin(); return; } originalOpenPoll(index); }; /* PROTECT HOME POLL */ const originalOpenPollFromHome = openPollFromHome; openPollFromHome=function(pollId){ if( !localStorage.getItem("didgah_remembered_user") && !sessionStorage.getItem("didgah_session_user") ){ showLogin(); return; } originalOpenPollFromHome(pollId); };
